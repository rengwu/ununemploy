import { mkdtempSync, rmSync, writeFileSync } from 'node:fs';
import { tmpdir } from 'node:os';
import { join } from 'node:path';
const directory=mkdtempSync(join(tmpdir(),'ununemploy-integration-audit-'));
process.env.JH_DATA_DIR=directory;
process.env.JH_PROFILE_SEED=new URL('../../../tests/fixtures/profile.json',import.meta.url).pathname;
const store=await import('../../../core/db/index.ts');
const q=await import('../../../core/boards/queue.ts');
const p=await import('../../../core/boards/parsers.ts');
const {collectPublicBoard}=await import('../../../core/boards/collector.ts');
const board=store.listSources().find(s=>s.type==='public_board' && s.identifier==='tryremote')!;
const url='https://tryremote.com/remote-jobs/123-audit-engineer';
const full={...p.lead('tryremote',url,'Engineer'),company:'Example',description:'Build accessible TypeScript applications. '.repeat(10),descriptionQuality:'full' as const,salary:'100',descriptionSource:{name:board.name,url,checkedAt:'2026-10-07T00:00:00Z'}};
const results=[];
try {
  store.ingestJob(board,full);
  const repeat=store.ingestJob(board,{...full,descriptionSource:{...full.descriptionSource,checkedAt:'2026-10-08T00:00:00Z'}});
  const versions=store.db.prepare('SELECT count(*) AS count FROM source_item_versions WHERE source_id=? AND source_item_id=?').get(board.id,full.sourceItemId)!.count;
  results.push({scenario:'unchanged content with new observation timestamp',expected:'updated=0 and one content version',observed:{updated:repeat.updated,versions},result:repeat.updated===0&&versions===1?'Passed':'Failed'});

  q.enqueueBoardJob(board.id,full);
  const updated=q.enqueueBoardJob(board.id,{...p.lead('tryremote',url,'Senior engineer'),company:'Example',salary:'200'});
  results.push({scenario:'new listing metadata retains full detail text',expected:'title Senior engineer; salary 200; original full description retained',observed:{title:updated.title,salary:updated.salary,fullDescriptionRetained:updated.description===full.description},result:updated.title==='Senior engineer'&&updated.salary==='200'&&updated.description===full.description?'Passed':'Failed'});

  q.failBoardJob(board.id,url,'HTTP 404',0,true);
  q.enqueueBoardJob(board.id,p.lead('tryremote',url,'Reopened engineer'));
  const queued=store.db.prepare('SELECT state FROM board_queue WHERE source_id=? AND url=?').get(board.id,url)!.state;
  const due=q.dueBoardJobs(board.id).some(r=>r.url===url);
  results.push({scenario:'previously closed URL reappears in discovery',expected:'eligible for bounded revalidation, without marking vacancy open yet',observed:{state:queued,due},result:due?'Passed':'Failed'});

  const html=`<script type="application/ld+json">${JSON.stringify({'@type':'JobPosting',url,title:'Engineer',hiringOrganization:{name:'Example'},description:'A short excerpt...',jobLocationType:'TELECOMMUTE',jobLocation:{address:{addressCountry:'Malaysia'}},applicantLocationRequirements:{'@type':'Country',name:'United States'}})}</script><section class="job-description">${full.description}</section>`;
  const detail=p.parseBoardDetail('tryremote',html,p.lead('tryremote',url));
  results.push({scenario:'visible full description preserves schema hiring restriction',expected:'full text plus United States applicant restriction; office remains Malaysia',observed:{quality:detail.descriptionQuality,location:detail.location,applicantRestrictionPresent:detail.description.includes('United States')},result:detail.description.includes('United States')?'Passed':'Failed'});

  let details=0;
  const queuedUrl='https://tryremote.com/remote-jobs/456-pending-engineer';
  q.enqueueBoardJob(board.id,p.lead('tryremote',queuedUrl,'Engineer'));
  let error='';
  try {await collectPublicBoard(board,{requests:0,limit:3},()=>{},()=>{},async (request,ctx)=>{ctx.requests++;if(request===queuedUrl){details++;return html;}return '<html><body>No readable listings</body></html>';});} catch(e){error=(e as Error).message;}
  results.push({scenario:'discovery shell prevents independent queued detail processing',classification:'design coupling; no new expected behavior asserted',observed:{detailRequests:details,seedError:error,pending:q.dueBoardJobs(board.id).length}});
  writeFileSync(new URL('./reproductions.json',import.meta.url),JSON.stringify(results,null,2)+'\n');
  console.log(JSON.stringify(results,null,2));
} finally {store.db.close();rmSync(directory,{recursive:true,force:true});}
