import { DatabaseSync } from 'node:sqlite';
import { writeFileSync } from 'node:fs';
import { createHash } from 'node:crypto';
import { config } from '../../../core/config.ts';
import { publicBoards } from '../../../core/boards/catalog.ts';
import { parseBoardListing, parseBoardDetail, jobUrl, lead } from '../../../core/boards/parsers.ts';
const db = new DatabaseSync(`${config.dataDir}/job-hunter.db`, { readOnly: true });
const sources = db.prepare('SELECT id,document FROM sources').all().map(r => JSON.parse(String(r.document)));
const caches = db.prepare('SELECT url,body,checked_at FROM http_cache').all();
const cached = new Map(caches.map(r => [String(r.url), r]));
const totals = Object.fromEntries(['sources','jobs','source_items','source_item_versions','board_queue','http_cache'].map(t => [t, db.prepare(`SELECT count(*) AS count FROM ${t}`).get()!.count]));
const byType = Object.fromEntries([...new Set(sources.map(s=>s.type))].map(t => [t, sources.filter(s=>s.type===t).length]));
const replay = [];
for (const [key, board] of Object.entries(publicBoards)) {
  const r = { key, setup: !!board.setup, listingPages: 0, listings: 0, listingErrors: 0, detailPages: 0, full: 0, excerpt: 0, detailErrors: 0, earliest: '', latest: '' };
  for(const [url,row] of cached) {
    if (board.seeds.some(seed => { const a=new URL(seed), b=new URL(url); return a.origin===b.origin && a.pathname===b.pathname; })) {
      r.listingPages++;
      try { r.listings += parseBoardListing(key,String(row.body),url).jobs.length; } catch { r.listingErrors++; }
    } else if(jobUrl(key,url)===url) {
      r.detailPages++;
      try { const parsed=parseBoardDetail(key,String(row.body),lead(key,url)); if(parsed.descriptionQuality==='full')r.full++;else r.excerpt++; } catch { r.detailErrors++; }
    } else continue;
    const d=String(row.checked_at).slice(0,10); r.earliest=!r.earliest||d<r.earliest?d:r.earliest; r.latest=d>r.latest?d:r.latest;
  }
  replay.push(r);
}
const versions = db.prepare('SELECT source_id,source_item_id,document FROM source_item_versions').all();
const fingerprints = new Set<string>();
let withCheckedAt=0;
for(const row of versions) {
  const item=JSON.parse(String(row.document));
  if(item.descriptionSource?.checkedAt) { withCheckedAt++; delete item.descriptionSource.checkedAt; }
  fingerprints.add(`${row.source_id}:${row.source_item_id}:${createHash('sha256').update(JSON.stringify(item)).digest('hex')}`);
}
const report = {
  collectedAt:new Date().toISOString(), databaseAccess:'read-only; no core/db import', totals, byType,
  states: Object.fromEntries([...new Set(sources.map(s=>s.state))].map(t=>[t,sources.filter(s=>s.state===t).length])),
  cacheBytes: caches.reduce((n,r)=>n+Buffer.byteLength(String(r.body)),0),
  queue: db.prepare('SELECT state,count(*) AS count FROM board_queue GROUP BY state').all(),
  versions:{total:versions.length,withCheckedAt,distinctIgnoringCheckedAt:fingerprints.size,duplicatesIgnoringCheckedAt:versions.length-fingerprints.size},
  cacheReplay:replay,
};
writeFileSync(new URL('./inventory.json',import.meta.url),JSON.stringify(report,null,2)+'\n');
console.log(JSON.stringify(report,null,2));
db.close();
