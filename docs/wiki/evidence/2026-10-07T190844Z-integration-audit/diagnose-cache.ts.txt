import { DatabaseSync } from 'node:sqlite';
import { writeFileSync } from 'node:fs';
import { parseDocument } from 'htmlparser2';
import { getElementsByTagName, textContent } from 'domutils';
import { config } from '../../../core/config.ts';
import { jobUrl, lead, parseBoardDetail } from '../../../core/boards/parsers.ts';
const db=new DatabaseSync(`${config.dataDir}/job-hunter.db`,{readOnly:true});
const failures: Record<string,Record<string,number>>={};
for(const row of db.prepare('SELECT url,body FROM http_cache').all()) {
  const url=String(row.url),html=String(row.body);
  for(const key of ['linkedin','nofluff','tryremote','remotegamejobs','glints']) {
    if(jobUrl(key,url)!==url)continue;
    try {parseBoardDetail(key,html,lead(key,url));continue;}catch{}
    const root=parseDocument(html);
    const postings:any[]=[];
    const visit=(v:any)=>{for(const o of (Array.isArray(v)?v:[v])){if(o&&typeof o==='object'){if([o['@type']].flat().includes('JobPosting'))postings.push(o);else if(o['@graph'])visit(o['@graph']);}}};
    for(const s of getElementsByTagName('script',root.children).filter(s=>s.attribs.type==='application/ld+json')){try{visit(JSON.parse(textContent(s)));}catch{}}
    const canonical=getElementsByTagName('link',root.children).find(s=>s.attribs.rel==='canonical')?.attribs.href;
    let reason='identity/body mismatch; requires fixture review';
    if(!postings.length)reason='no parseable JobPosting in response';
    else if(postings.length===1&&!postings[0].url&&!postings[0]['@id']&&canonical&&jobUrl(key,canonical)!==url)reason=key==='glints'&&canonical.endsWith('/opportunities/jobs/id')?'generic /opportunities/jobs/id canonical; no schema URL':'different canonical; no schema URL';
    failures[key]??={};failures[key][reason]=(failures[key][reason]||0)+1;
  }
}
const backlog=db.prepare(`SELECT s.identifier, q.state,count(*) AS count,
 sum(CASE WHEN q.state='full' AND q.checked_at<datetime('now','-7 days') THEN 1 ELSE 0 END) AS fullOlderThanSevenDays
 FROM board_queue q JOIN sources s ON s.id=q.source_id GROUP BY s.identifier,q.state`).all();
const result={failures,backlog};
writeFileSync(new URL('./cache-diagnosis.json',import.meta.url),JSON.stringify(result,null,2)+'\n');
console.log(JSON.stringify(failures,null,2));db.close();
