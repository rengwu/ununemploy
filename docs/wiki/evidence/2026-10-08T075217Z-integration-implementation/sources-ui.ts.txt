import assert from 'node:assert/strict';
import { mkdtempSync,rmSync,writeFileSync } from 'node:fs';
import { tmpdir } from 'node:os';
import { join } from 'node:path';
import { chromium } from 'playwright';
const directory=mkdtempSync(join(tmpdir(),'ununemploy-sources-ui-'));
process.env.JH_DATA_DIR=directory;
process.env.JH_PROFILE_SEED=new URL('../../../tests/fixtures/profile.json',import.meta.url).pathname;
process.env.JH_TEST_SERVER='1';
const store=await import('../../../core/db/index.ts');
store.putSetting('app',{...store.getSettings(),scheduleEnabled:false,aiEnabled:false,xEnabled:false});
const {app}=await import('../../../server/index.ts');
const origin=await app.listen({host:'127.0.0.1',port:0});
const browser=await chromium.launch();
const page=await browser.newPage({viewport:{width:1440,height:1000}});
const errors:string[]=[];page.on('pageerror',e=>errors.push(e.message));
let calls=0;
const original=globalThis.fetch;
globalThis.fetch=async(url)=>{ assert.equal(String(url),'https://www.workingnomads.com/api/exposed_jobs/'); calls++; return new Response(JSON.stringify([{url:'https://www.workingnomads.com/job/go/123/',title:'Product Engineer',company_name:'Example',description:'Build accessible TypeScript software and collaborate on application design. '.repeat(5)}]));};
try{
  await page.goto(origin+'/#sources');
  const names=['Working Nomads','WordPress Jobs','RemoteJobs.com','Dynamite Jobs'];
  for(const name of names){
    const button=page.getByRole('button',{name,exact:true});await button.waitFor();assert.equal(await button.count(),1);
    await button.click();await page.getByRole('dialog').waitFor();
    assert.equal(await page.getByRole('textbox',{name:'Name',exact:true}).inputValue(),name);
    await page.getByRole('button',{name:'Cancel',exact:true}).click();
  }
  const row=page.getByRole('row').filter({has:page.getByRole('button',{name:'Working Nomads',exact:true})});
  const queued=page.waitForResponse(r=>r.url()===origin+'/api/scan');
  await row.getByRole('button',{name:'Scan',exact:true}).click();
  assert.equal((await queued).status(),202);
  const source=store.listSources().find(s=>s.type==='workingnomads')!;
  const deadline=Date.now()+10000;
  while(store.getSource(source.id)?.state!=='idle' || !store.getSource(source.id)?.lastRunAt){
    if(Date.now()>deadline)throw new Error('Scan did not finish');
    await new Promise(r=>setTimeout(r,100));
  }
  assert.equal(calls,1);assert.equal(store.listJobs().length,1);assert.equal(store.listJobs()[0].descriptionQuality,'full');
  assert.ok(!store.listSources().some(s=>/the muse|reddit/i.test(s.name)));
  assert.deepEqual(errors,[]);
  const result={at:new Date().toISOString(),sourceControls:names,sourceCount:store.listSources().length,uiScan:{status:202,providerRequests:calls,persistedJobs:1,fullDescription:true},browserErrors:errors,isolated:true};
  writeFileSync(new URL('./sources-ui.json',import.meta.url),JSON.stringify(result,null,2)+'\n');
  console.log(JSON.stringify(result));
}finally{globalThis.fetch=original;await browser.close();await app.close();store.db.close();rmSync(directory,{recursive:true,force:true});}
