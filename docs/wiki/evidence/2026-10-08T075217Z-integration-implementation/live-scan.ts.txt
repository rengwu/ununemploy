import { mkdtempSync, rmSync, writeFileSync } from 'node:fs';
import { tmpdir } from 'node:os';
import { join } from 'node:path';
const dir=mkdtempSync(join(tmpdir(),'ununemploy-implementation-live-'));
process.env.JH_DATA_DIR=dir;
process.env.JH_PROFILE_SEED=new URL('../../../tests/fixtures/profile.json',import.meta.url).pathname;
const store=await import('../../../core/db/index.ts');
const {scan}=await import('../../../core/scan.ts');
const selections=[
  {type:'workingnomads',identifier:'workingnomads',limit:1},
  {type:'rss',identifier:'https://jobs.wordpress.net/feed/',limit:1},
  {type:'rss',identifier:'https://www.remotejobs.com/jobs.rss',limit:1},
  {type:'public_board',identifier:'dynamite',limit:3},
  {type:'ashby',identifier:'langchain',limit:1},
  {type:'greenhouse',identifier:'figma',limit:1},
  {type:'lever',identifier:'palantir',limit:1},
  {type:'rss',identifier:'https://weworkremotely.com/categories/remote-programming-jobs.rss',limit:1},
  {type:'remoteok',identifier:'remoteok',limit:1},
  {type:'himalayas',identifier:'frontend',limit:1},
  {type:'fourdayweek',identifier:'engineering,product,design,data',limit:1},
  {type:'hn_jobs',identifier:'jobstories',limit:2},
  {type:'public_board',identifier:'tryremote',limit:2},
  {type:'public_board',identifier:'getonboard',limit:1},
];
const reports=[];
try {
  for(const selection of selections) {
    let source=store.listSources().find(s=>s.type===selection.type && (s.identifier===selection.identifier || s.type==='himalayas'));
    if(!source)source=store.addSource({name:`Audit ${selection.type}`,type:selection.type as any,identifier:selection.identifier,requestLimit:selection.limit,enabled:true});
    source=store.saveSource({...source,identifier:selection.identifier,requestLimit:selection.limit,enabled:true});
    const start=Date.now();
    const response=await scan(source.id,true);
    const run=response.runs?.[0];
    const jobs=store.db.prepare('SELECT document FROM source_items WHERE source_id=?').all(source.id).map(r=>JSON.parse(String(r.document)));
    const invalid=jobs.filter(j=>!j.sourceItemId||!j.title||!/^https?:/.test(j.url)||typeof j.description!=='string').length;
    const r={type:selection.type,identifier:selection.identifier,at:new Date().toISOString(),durationMs:Date.now()-start,status:run?.status,requests:run?.requests,added:run?.added,error:run?.error,stored:jobs.length,full:jobs.filter(j=>j.descriptionQuality==='full').length,empty:jobs.filter(j=>!j.description.trim()).length,invalidIdentityOrDescription:invalid,checkpointPreserved:Object.keys(store.getSource(source.id)!.checkpoint).length>0};
    reports.push(r); console.log(JSON.stringify(r));
  }
  writeFileSync(new URL('./live-scan.json',import.meta.url),JSON.stringify(reports,null,2)+'\n');
}finally{store.db.close();rmSync(dir,{recursive:true,force:true});}
