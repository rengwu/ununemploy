import { DatabaseSync } from 'node:sqlite';
import { writeFileSync } from 'node:fs';
import { createHash } from 'node:crypto';
import { config } from '../../../core/config.ts';
import { publicBoards } from '../../../core/boards/catalog.ts';
import * as current from '../../../core/boards/parsers.ts';
import * as baseline from './baseline-parser.ts';
const db = new DatabaseSync(`${config.dataDir}/job-hunter.db`, { readOnly: true });
const caches = db.prepare('SELECT url,body FROM http_cache').all();
db.close();
const fingerprint=createHash('sha256');
const report={ at:new Date().toISOString(), databaseAccess:'read-only', baseline:'git HEAD core/boards/parsers.ts with import paths adjusted', listingPages:0, detailPages:0, identicalListings:0, identicalDetails:0, retainedRestrictions:0, sameRejections:0, differences:[] as unknown[], corpusFingerprint:'' };
function parse(fn:()=>unknown){try{return {ok:true,data:fn()};}catch(error){return {ok:false,error:error instanceof Error?error.message:String(error)};}}
for(const [key,board] of Object.entries(publicBoards)){
  if(key==='dynamite')continue;
  for(const row of caches){
    const url=String(row.url),body=String(row.body);
    const listing=board.seeds.some(seed=>{const a=new URL(seed),b=new URL(url);return a.origin===b.origin&&a.pathname===b.pathname;});
    if(!listing&&current.jobUrl(key,url)!==url)continue;
    fingerprint.update(key).update(url).update(body);
    const old=parse(()=>listing?baseline.parseBoardListing(key,body,url):baseline.parseBoardDetail(key,body,baseline.lead(key,url)));
    const next=parse(()=>listing?current.parseBoardListing(key,body,url):current.parseBoardDetail(key,body,current.lead(key,url)));
    if(listing)report.listingPages++;else report.detailPages++;
    if(JSON.stringify(old)===JSON.stringify(next)){if(!old.ok)report.sameRejections++;else if(listing)report.identicalListings++;else report.identicalDetails++;continue;}
    const a=old.data as any,b=next.data as any;
    if(!listing&&old.ok&&next.ok&&b.description.startsWith(a.description+'\n\nApplicant locations:')&&JSON.stringify({...a,description:''})===JSON.stringify({...b,description:''})) {report.retainedRestrictions++;continue;}
    report.differences.push({key,url,old:old.ok?'parsed':old.error,new:next.ok?'parsed':next.error});
  }
}
report.corpusFingerprint=fingerprint.digest('hex');
writeFileSync(new URL('./cache-comparison.json',import.meta.url),JSON.stringify(report,null,2)+'\n');
console.log(JSON.stringify(report));
if(report.differences.length)process.exitCode=1;
