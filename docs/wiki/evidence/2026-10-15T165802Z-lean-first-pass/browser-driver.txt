import assert from "node:assert/strict";
import { mkdtempSync, rmSync, writeFileSync } from "node:fs";
import { tmpdir } from "node:os";
import { join, resolve } from "node:path";
import { pathToFileURL } from "node:url";
import { randomUUID } from "node:crypto";
import { chromium } from "playwright";
const directory = mkdtempSync(join(tmpdir(), "ununemploy-lean-browser-"));
process.env.JH_DATA_DIR = directory;
process.env.JH_PROFILE_SEED = resolve("tests/fixtures/profile.json");
process.env.JH_TEST_SERVER = "1";
process.env.JH_SCHEDULE_ENABLED = "false";
const store = await import(pathToFileURL(resolve("core/db/index.ts")).href);
store.putSetting("app", { ...store.getSettings(), scheduleEnabled: false, aiEnabled: false });
for (const source of store.listSources()) store.saveSource({ ...source, enabled: false });
for (const identifier of ["adzuna", "jooble", "careerjet"]) {
 const source = { ...store.listSources()[0], id: randomUUID(), type: "public_board", identifier, name: identifier, enabled: true };
 store.db.prepare("INSERT INTO sources VALUES (?,?,?,?)").run(source.id, JSON.stringify(source), source.type, identifier);
}
const { app } = await import(pathToFileURL(resolve("server/index.ts")).href);
let browser;
const errors: string[] = [];
try {
 const origin = await app.listen({ host: "127.0.0.1", port: 0 });
 browser = await chromium.launch({ headless: true });
 const page = await browser.newPage();
 page.on("pageerror", error => errors.push(error.message));
 await page.addInitScript(() => {
   (window as any).__registeredTools = [];
   Object.defineProperty(document, "modelContext", { value: { registerTool(tool: any) { (window as any).__registeredTools.push(tool.name); } } });
 });
 await page.goto(origin + "/#sources");
 await page.getByRole("button", { name: "Add source", exact: true }).waitFor();
 const sourceText = await page.locator("main").innerText();
 assert.doesNotMatch(sourceText, /adzuna|jooble|careerjet/i);
 assert.match(sourceText, /Hiredly/);
 await page.getByRole("button", { name: "Add source", exact: true }).click();
 await page.getByRole("combobox", { name: "Source type" }).click();
 await page.getByRole("option", { name: "Public job board", exact: true }).click();
 await page.getByRole("combobox", { name: "Board", exact: true }).click();
 await page.getByRole("option", { name: "Hiredly", exact: true }).waitFor();
 const options = await page.getByRole("option").allTextContents();
 assert.ok(options.some(x => x.includes("Hiredly")));
 assert.ok(options.every(x => !/adzuna|jooble|careerjet/i.test(x)));
 await page.keyboard.press("Escape");
 await page.keyboard.press("Escape");
 await page.goto(origin + "/#jobs");
 await page.getByRole("button", { name: "Add job", exact: true }).waitFor();
 assert.deepEqual(await page.evaluate(() => (window as any).__registeredTools), []);
 assert.deepEqual(errors, []);
 const result = { result: "Passed", scenarios: ["Sources hides retained placeholders and shows supported boards", "Add source offers supported boards only", "Jobs UI loads with no WebMCP registrations", "No browser page errors"], retainedPlaceholderRows: 3, isolated: true };
 writeFileSync(resolve(".local-data/verification/lean-pass1-browser.json"), JSON.stringify(result, null, 2));
 console.log(JSON.stringify(result));
} finally {
 await browser?.close();
 await app.close();
 store.db.close();
 rmSync(directory, { recursive: true, force: true });
}
