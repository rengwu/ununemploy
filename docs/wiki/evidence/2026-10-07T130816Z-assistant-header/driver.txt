import assert from 'node:assert/strict';
import { mkdtempSync, rmSync, writeFileSync } from 'node:fs';
import { tmpdir } from 'node:os';
import { join, resolve } from 'node:path';
import { chromium } from 'playwright';
const data = mkdtempSync(join(tmpdir(), 'ununemploy-header-'));
process.env.JH_DATA_DIR = data;
process.env.JH_PROFILE_SEED = resolve('tests/fixtures/profile.json');
process.env.JH_TEST_SERVER = '1';
process.env.JH_SCHEDULE_ENABLED = 'false';
const store = await import('../../core/db/index.ts');
for (const source of store.listSources()) store.saveSource({ ...source, enabled: false });
const { app } = await import('../../server/index.ts');
const origin = await app.listen({ host: '127.0.0.1', port: 0 });
const browser = await chromium.launch();
const page = await browser.newPage();
const results = [];
const errors: string[] = [];
page.on('pageerror', e => errors.push(e.message));
try {
  await page.goto(origin + '/#jobs');
  await page.getByRole('button', { name: 'Open assistant', exact: true }).click();
  await page.getByRole('textbox', { name: 'Message assistant', exact: true }).waitFor();
  for (const viewport of [{ width: 1440, height: 900 }, { width: 900, height: 540 }, { width: 390, height: 844 }]) {
    await page.setViewportSize(viewport);
    for (const theme of ['light', 'dark']) {
      await page.evaluate(theme => { document.documentElement.dataset.mode = theme; }, theme);
      const bounds: Record<string, { top: number; bottom: number; height: number }> = {};
      for (const [key, selector] of Object.entries({ main: '.page-header', assistant: '.assistant-header', actions: '.assistant-header-actions' })) {
        const r = await page.locator(selector).boundingBox();
        assert.ok(r);
        bounds[key] = { top: r.y, bottom: r.y + r.height, height: r.height };
      }
      assert.deepEqual(bounds.assistant, bounds.main);
      assert.equal(bounds.assistant.height, 58);
      assert.ok(bounds.actions.top >= bounds.assistant.top && bounds.actions.bottom <= bounds.assistant.bottom);
      results.push({ viewport, theme, ...bounds });
    }
  }
  assert.deepEqual(errors, []);
  writeFileSync('.local-data/verification/assistant-header-results.json', JSON.stringify({ passed: true, results, errors }, null, 2) + '\n');
  console.log('Passed: matching 58px headers and contained controls at three sizes in light and dark themes.');
} finally {
  await browser.close();
  await app.close();
  store.db.close();
  rmSync(data, { recursive: true, force: true });
}
