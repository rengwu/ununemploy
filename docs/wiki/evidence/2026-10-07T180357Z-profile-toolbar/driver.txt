import assert from 'node:assert/strict';
import { mkdtempSync, rmSync, writeFileSync } from 'node:fs';
import { tmpdir } from 'node:os';
import { join, resolve } from 'node:path';
import { chromium } from 'playwright';
const data = mkdtempSync(join(tmpdir(), 'ununemploy-profile-toolbar-'));
process.env.JH_DATA_DIR = data;
process.env.JH_PROFILE_SEED = resolve('tests/fixtures/profile.json');
process.env.JH_TEST_SERVER = '1';
const store = await import('../../core/db/index.ts');
const { emptyProfile } = await import('../../core/profile/facts.ts');
for (const source of store.listSources()) store.saveSource({ ...source, enabled: false });
store.putSetting('app', { ...store.getSettings(), scheduleEnabled: true, aiEnabled: false });
const original = store.getProfile();
store.saveProfile({ ...original, basics: { ...original.basics, name: 'Toolbar Example' } }, original.revision);
const { app } = await import('../../server/index.ts');
const origin = await app.listen({ host: '127.0.0.1', port: 0 });
const browser = await chromium.launch();
const page = await browser.newPage({ viewport: { width: 1440, height: 900 } });
const errors: string[] = [];
page.on('pageerror', e => errors.push(e.message));
const layouts = [];
try {
  await page.goto(origin + '/#profile');
  const select = page.getByRole('combobox', { name: 'Profile version to restore', exact: true });
  const restore = page.getByRole('button', { name: 'Restore version', exact: true });
  await select.waitFor();
  assert.equal(await restore.isDisabled(), true);
  assert.equal(await page.locator('footer.profile-version-actions').count(), 0);
  assert.equal(await page.getByRole('heading', { name: 'Profile facts', exact: true }).count(), 0);
  for (const viewport of [{ width: 1440, height: 900 }, { width: 390, height: 844 }]) {
    await page.setViewportSize(viewport);
    await page.waitForFunction(() => innerWidth >= 768 || Math.round(document.querySelector('.app-main')!.getBoundingClientRect().width) === innerWidth);
    for (const theme of ['light', 'dark']) {
      await page.evaluate(theme => { document.documentElement.dataset.mode = theme; }, theme);
      const toolbar = await page.locator('.profile-toolbar').boundingBox();
      const status = await page.locator('.profile-readiness').boundingBox();
      const edit = await page.getByRole('button', { name: 'Edit with assistant', exact: true }).boundingBox();
      const versions = await page.locator('.profile-version-actions').boundingBox();
      const facts = await page.locator('.profile-facts-scroll').boundingBox();
      assert.ok(toolbar && status && edit && versions && facts);
      assert.ok(edit.y >= toolbar.y && edit.y + edit.height <= toolbar.y + toolbar.height);
      assert.ok(versions.y >= toolbar.y && versions.y + versions.height <= toolbar.y + toolbar.height);
      assert.equal(facts.y, toolbar.y + toolbar.height);
      if (viewport.width === 1440) assert.ok(Math.abs(status.y + status.height / 2 - edit.y - edit.height / 2) < 1);
      assert.equal(await page.evaluate(() => document.documentElement.scrollWidth), viewport.width);
      layouts.push({ viewport, theme, toolbar, status, edit, versions, facts });
      if (theme === 'light') await page.screenshot({ path: `.local-data/verification/profile-toolbar-${viewport.width}.png` });
    }
  }
  await select.click();
  await page.getByRole('option', { name: new RegExp(`^Revision ${original.revision} ·`) }).click();
  assert.equal(await restore.isDisabled(), false);
  const restored = page.waitForResponse(r => r.url().endsWith('/api/profile/restore') && r.request().method() === 'POST');
  await restore.click();
  assert.equal((await restored).status(), 200);
  assert.equal(store.getProfile().basics.name, original.basics.name);
  assert.equal(store.getProfile().revision, original.revision + 2);
  await page.getByRole('button', { name: 'Edit with assistant', exact: true }).click();
  await page.getByRole('textbox', { name: 'Message assistant', exact: true }).waitFor();
  await page.getByRole('button', { name: 'Hide assistant', exact: true }).click();
  store.putSetting('app', { ...store.getSettings(), scheduleEnabled: false });
  await page.reload();
  await page.getByText('Profile ready. Scheduled scanning is paused in Settings.', { exact: true }).waitFor();
  store.saveProfile(emptyProfile(), store.getProfile().revision);
  await page.reload();
  await page.getByRole('button', { name: 'Build profile', exact: true }).waitFor();
  await page.getByText('Job scanning is waiting for your profile.', { exact: true }).waitFor();
  assert.match(await page.locator('.profile-readiness').innerText(), /Still needed: name, current location, target roles, career background, work preferences/);
  assert.equal(await page.evaluate(() => document.documentElement.scrollWidth), 390);
  assert.deepEqual(errors, []);
  writeFileSync('.local-data/verification/profile-toolbar-results.json', JSON.stringify({ passed: true, layouts, errors, checks: ['single desktop toolbar; no redundant heading/footer', 'responsive light/dark layout without overflow', 'disabled restore until revision selected', 'mobile restore creates new revision', 'Edit with assistant opens composer', 'paused and incomplete readiness remains visible'] }, null, 2) + '\n');
  console.log('Passed: unified profile toolbar, desktop/mobile layout, restore, assistant action and readiness variants.');
} finally {
  await browser.close();
  await app.close();
  store.db.close();
  rmSync(data, { recursive: true, force: true });
}
